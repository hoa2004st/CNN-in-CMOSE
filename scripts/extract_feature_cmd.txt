# Filter unusable clips, write accepted.csv
@'
from pathlib import Path
import pandas as pd

clips = sorted(Path("data/private/clips").glob("*.mp4"))
rows = []
for clip in clips:
    clip_id = clip.stem
    csv_path = Path("data/private/features/openface") / f"{clip_id}.csv"
    ok, reason = 1, ""
    if not csv_path.exists():
        ok, reason = 0, "openface_csv_missing"
    else:
        try:
            df = pd.read_csv(csv_path)
            df.columns = df.columns.str.strip()
            if len(df) == 0:
                ok, reason = 0, "empty_csv"
            elif "success" not in df.columns:
                ok, reason = 0, "missing_success_col"
            else:
                succ = int((df["success"] > 0.5).sum())
                ratio = succ / max(len(df), 1)
                if succ < 30:
                    ok, reason = 0, "too_few_success_frames"
                elif ratio < 0.30:
                    ok, reason = 0, "low_success_ratio"
        except Exception:
            ok, reason = 0, "csv_parse_error"
    rows.append({
        "clip_id": clip_id,
        "clip_path": str(clip),
        "openface_csv": str(csv_path),
        "is_accepted": ok,
        "reject_reason": reason,
    })

out = Path("data/private/accepted.csv")
out.parent.mkdir(parents=True, exist_ok=True)
pd.DataFrame(rows).to_csv(out, index=False)
print(f"written={out} accepted={sum(r['is_accepted'] for r in rows)} total={len(rows)}")
'@ | python -

# Convert MMAction outputs to CMOSE-compatible 1024-dim .npy per clip
@'
import pickle
from pathlib import Path
import numpy as np

src_root = Path("artifacts/i3d/raw_features/data/private/clips")
out_root = Path("data/private/features/i3d")
out_root.mkdir(parents=True, exist_ok=True)

written = 0
for pkl_path in sorted(src_root.glob("*.mp4.pkl")):
    with pkl_path.open("rb") as f:
        obj = pickle.load(f)
    vec = np.asarray(obj, dtype=np.float32).reshape(-1)
    if vec.size == 2048:
        vec = vec.reshape(1024, 2).mean(axis=1).astype(np.float32)
    elif vec.size != 1024:
        raise ValueError(f"Unexpected feature dim {vec.size} in {pkl_path}")
    clip_id = pkl_path.name.replace(".mp4.pkl", "")
    np.save(out_root / f"{clip_id}.npy", vec)
    written += 1

print("written", written, "files to", out_root)
